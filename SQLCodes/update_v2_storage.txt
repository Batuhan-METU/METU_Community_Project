-- Görsel Yükleme (1.7) — Storage URL sütunları
-- profiles, communities ve events tablolarına resim URL'leri için TEXT sütunlar eklenir.

-- 1) profiles tablosuna profil fotoğrafı URL sütunu
ALTER TABLE public.profiles
ADD COLUMN IF NOT EXISTS avatar_url text;

-- 2) communities tablosuna topluluk logosu URL sütunu (yoksa ekler)
ALTER TABLE public.communities
ADD COLUMN IF NOT EXISTS logo_url text;

-- 3) events tablosuna etkinlik kapak resmi URL sütunu
ALTER TABLE public.events
ADD COLUMN IF NOT EXISTS image_url text;
